import { onMounted } from 'vue'

onMounted(() => {
  async function loadStats() {
    const el = document.getElementById('repoCount')
    if (!el) return
    try {
      const r = await fetch('https://api.github.com/users/ibqy')
      if (r.ok) {
        const d = await r.json()
        el.textContent = d.public_repos
      }
    } catch (_) {}
  }

  function escapeHtml(s) {
    const d = document.createElement('div')
    d.appendChild(document.createTextNode(s))
    return d.innerHTML
  }

  function plainPreview(body) {
    return (body || '')
      .split('\n')
      .filter((l) => !l.trim().startsWith('#'))
      .map((l) => l.replace(/^\s*[-*]\s+/, ''))
      .join(' ')
      .replace(/\*\*?|__?|`/g, '')
      .replace(/\s+/g, ' ')
      .trim()
      .slice(0, 120)
  }

  async function loadBlog() {
    const c = document.getElementById('blogContainer')
    if (!c) return
    try {
      const res = await fetch(
        'https://api.github.com/repos/ibqy/ibqy/issues?labels=blog&state=open&per_page=5&sort=created&direction=desc'
      )
      if (!res.ok) throw new Error('HTTP ' + res.status)
      const issues = await res.json()
      if (!issues.length) {
        c.innerHTML =
          '<div class="blog-empty"><p>还没有博客文章。在 GitHub Issues 中添加 blog 标签即可发布。</p></div>'
        return
      }
      c.innerHTML =
        '<div class="blog-list">' +
        issues
          .map(
            (i) =>
              '<a href="' +
              i.html_url +
              '" class="blog-card" target="_blank" rel="noopener">' +
              '<div class="blog-card-icon">📄</div>' +
              '<div>' +
              '<h4>' +
              escapeHtml(i.title) +
              '</h4>' +
              '<p>' +
              escapeHtml(plainPreview(i.body)) +
              '</p>' +
              '<div class="blog-date">' +
              new Date(i.created_at).toLocaleDateString('zh-CN') +
              '</div>' +
              '</div>' +
              '</a>'
          )
          .join('') +
        '</div>'
    } catch (_) {
      c.innerHTML =
        '<div class="blog-empty"><p>博客加载失败。前往 <a href="https://github.com/ibqy/ibqy/issues" style="color:var(--vp-c-brand-1)">GitHub Issues</a> 查看。</p></div>'
    }
  }

  function syncCounts() {
    const aiKeywords = [
      'Spring AI',
      'LLM',
      'RAG',
      '混合检索',
      'Rerank',
      'Agent',
      '多智能体',
      '编排',
      'Semantic Kernel',
      'AgentScope',
      '通义千问',
      'DashScope'
    ]
    const cards = document.querySelectorAll('.bento-card')
    let ai = 0
    cards.forEach((card) => {
      const tags = [...card.querySelectorAll('.tags span')].map((t) => t.textContent.trim())
      if (tags.some((t) => aiKeywords.some((k) => t.includes(k)))) ai++
    })
    const aiEl = document.getElementById('aiCount')
    const beEl = document.getElementById('backendCount')
    if (aiEl) aiEl.textContent = ai
    if (beEl) beEl.textContent = cards.length - ai
  }

  loadStats()
  loadBlog()
  syncCounts()
})
</script>
